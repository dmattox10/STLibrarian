# robots.txt for stlibrarian.com
#
# The page itself is static HTML with the whole pitch already in it, so it is
# one of the few sites here a crawler could read before this round of work.
# What it was missing was any way to be found: no sitemap, and a request for
# /robots.txt fell through nginx's SPA catch-all and came back as the homepage
# with a 200.
#
# This directory is bind-mounted straight into the container
# (/root/docker/docker-compose.yml, stlibrarian-site), so a change here is live
# after a git pull - there is no image to rebuild.
#
# tools/ is build tooling - capture.mjs, seed.mjs, make-models.mjs - that lives
# inside the document root and is therefore reachable over HTTP. Keeping it out
# of the index is the least that should happen to it; serving it at all is a
# separate question for nginx.
User-agent: *
Allow: /
Disallow: /tools/

Sitemap: https://stlibrarian.com/sitemap.xml
